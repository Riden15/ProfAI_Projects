============================================================
FASE 3 - ANALISI DEI TEMPI DI RICOVERO
============================================================


# Tempo medio GLOBALE di ricovero (in giorni)

pipeline_globale = [
    # Decomponi l'array "ricoveri" in documenti singoli
    { "$unwind": "$ricoveri" },

    # Calcola la durata in giorni per ogni ricovero
    {
        "$project": {
            "nome": 1,
            "durata_giorni": {
                "$divide": [
                    { "$subtract": ["$ricoveri.data_dimissione", "$ricoveri.data"] },
                    1000 * 60 * 60 * 24   # conversione da millisecondi a giorni
                ]
            }
        }
    },

    # Raggruppa tutti i ricoveri e calcola la media globale
    {
        "$group": {
            "_id": None,
            "media_giorni_globale": { "$avg": "$durata_giorni" },
            "totale_ricoveri": { "$sum": 1 }
        }
    },

    # Formatta il risultato
    {
        "$project": {
            "_id": 0,
            "media_giorni_globale": { "$round": ["$media_giorni_globale", 2] },
            "totale_ricoveri": 1
        }
    }
]

risultato_globale = list(pazienti.aggregate(pipeline_globale))

print("Tempo medio globale di ricovero (tutti i pazienti):")
print(f"  Totale ricoveri analizzati : {risultato_globale[0]['totale_ricoveri']}")
print(f"  Media giorni di degenza    : {risultato_globale[0]['media_giorni_globale']} giorni")


# Tempo medio di ricovero PER PAZIENTE

pipeline_per_paziente = [
    { "$unwind": "$ricoveri" },

    {
        "$project": {
            "nome": 1,
            "durata_giorni": {
                "$divide": [
                    { "$subtract": ["$ricoveri.data_dimissione", "$ricoveri.data"] },
                    1000 * 60 * 60 * 24
                ]
            }
        }
    },

    {
        "$group": {
            "_id": { "id": "$_id", "nome": "$nome" },
            "media_giorni": { "$avg": "$durata_giorni" },
            "num_ricoveri": { "$sum": 1 }
        }
    },

    {
        "$project": {
            "_id": 0,
            "nome": "$_id.nome",
            "num_ricoveri": 1,
            "media_giorni": { "$round": ["$media_giorni", 2] }
        }
    },

    { "$sort": { "nome": 1 } }
]

risultato_per_paziente = list(pazienti.aggregate(pipeline_per_paziente))

for p in risultato_per_paziente:
    print(
        f"Paziente: {p['nome']} "
        f"| Ricoveri: {p['num_ricoveri']} "
        f"| Media degenza: {p['media_giorni']} giorni"
    )

client.close()
